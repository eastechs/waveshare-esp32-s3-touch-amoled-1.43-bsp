cmake_minimum_required(VERSION 3.20)
project(waveshare_esp32_s3_touch_amoled_143_bsp_tests LANGUAGES C)
include(CTest)
if(BUILD_TESTING)
    foreach(name IN ITEMS shake changed_frame)
        add_executable(test-${name} tests/${name}.c board/${name}.c)
        target_include_directories(test-${name} PRIVATE board)
        target_compile_features(test-${name} PRIVATE c_std_11)
        target_compile_options(test-${name} PRIVATE -UNDEBUG -Wall -Wextra -Werror)
        target_link_libraries(test-${name} PRIVATE m)
        add_test(NAME ${name} COMMAND test-${name})
    endforeach()
endif()
